                tab.classList.add('bg-white', 'text-slate-700', 'border', 'border-slate-200');

            if(activeTab) {

                activeTab.classList.add('bg-emerald-600', 'text-white');

            // Filter cards

                    card.style.display = 'block';
                } else {
                    card.style.display = 'none';
